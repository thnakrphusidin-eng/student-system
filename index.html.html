● พร้อมใช้งาน

เลือกใบงานที่ต้องการทดลอง กรอกข้อมูล แล้วกดปุ่มเพื่อดูผลลัพธ์

โปรแกรมตรวจสอบผลการสอบ

กำหนดคะแนนตั้งแต่ 50 คะแนนขึ้นไปเป็น PASS

If - Else
ผลลัพธ์จะแสดงที่นี่
int score = 75;

if (score >= 50) {
    System.out.println("Result: PASS");
} else {
    System.out.println("Result: FAIL");
}

โปรแกรมคำนวณระดับผลการเรียน

แปลผลคะแนนตามเกณฑ์ระดับ A ถึง F

Else If
ผลลัพธ์จะแสดงที่นี่

เกณฑ์: 80–100 A · 75–79 B+ · 70–74 B · 65–69 C+ · 60–64 C · 55–59 D+ · 50–54 D · 0–49 F

int score = 85;
String grade;

if (score >= 80) grade = "A";
else if (score >= 75) grade = "B+";
else if (score >= 70) grade = "B";
else if (score >= 65) grade = "C+";
else if (score >= 60) grade = "C";
else if (score >= 55) grade = "D+";
else if (score >= 50) grade = "D";
else grade = "F";

System.out.println("Grade: " + grade);

โปรแกรมตรวจสอบสิทธิ์ส่วนลด

ซื้อสินค้าตั้งแต่ 1,000 บาทขึ้นไป รับส่วนลด 10%

Ternary Operator
ผลลัพธ์จะแสดงที่นี่
double price = 1500.0;
boolean eligible = price >= 1000;
double discount = eligible ? price * 0.10 : 0;
double netPrice = price - discount;

System.out.println("Discount: " + discount);
System.out.println("Net Price: " + netPrice);

โปรแกรมตรวจสอบสิทธิ์เข้าห้องปฏิบัติการ

ต้องมีบัตรและเป็นนักเรียนปัจจุบัน พร้อมได้รับอนุมัติจากครูหรือเป็นผู้ดูแลแล็บ

Nested If + Logical
ผลลัพธ์จะแสดงที่นี่
if (hasStudentCard && isActiveStudent) {
    if (isTeacherApproved || isLabAdmin) {
        System.out.println("Access Granted");
    } else {
        System.out.println("Access Denied");
    }
} else {
    System.out.println("Access Denied");
}

โปรแกรมระบบเลือกเมนูบริการ

เลือกหมายเลขเมนูเพื่อดูรายการบริการ

Switch Case
1. Student Information — ตรวจสอบข้อมูลนักเรียน
2. Exam Scores — ตรวจสอบคะแนนสอบ
3. Class Schedule — ตรวจสอบตารางเรียน
4. Contact Teacher — ติดต่อครูผู้สอน
5. Exit — ออกจากระบบ
ผลลัพธ์จะแสดงที่นี่
int menu = 3;

switch (menu) {
    case 1:
        System.out.println("You selected Student Information.");
        break;
    case 2:
        System.out.println("You selected Exam Scores.");
        break;
    case 3:
        System.out.println("You selected Class Schedule.");
        break;
    case 4:
        System.out.println("You selected Contact Teacher.");
        break;
    case 5:
        System.out.println("You selected Exit.");
        break;
    default:
        System.out.println("Invalid Menu.");
}
